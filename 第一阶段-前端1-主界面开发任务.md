## 〇、相对 v1 的关键变更

| 维度 | v1（旧版，错的） | v2（本版，对齐后端） |
| --- | --- | --- |
| 后端技术栈 | 假设是 Kotlin / Room | **Python 3 + sqlite3，HTTP 8800 端口** |
| 数据存储 | Android 端 Room 本地库 | **Python 维护的 sqlite，Android 通过 HTTP 访问** |
| ID 类型 | `String` / UUID | **`Long`**（后端数据库自增） |
| 字段命名 | Kotlin camelCase | **camelCase + `@SerialName` 映射后端 snake_case** |
| 返回值结构 | 直接 data class | **统一 `ApiResponse<T>` 包装**（后端 `{ok, data, error}`） |
| HTTP 客户端 | 没考虑 | **Ktor Client**（Kotlin 原生 + 协程） |
| JSON 序列化 | 没考虑 | **kotlinx-serialization**（Kotlin 官方） |
| 冲突显示 | 推迟到第二阶段 | **第一阶段 P0 就要做**（后端已返回 items 含 conflict_pair/conflict_group） |
| Reminder 字段 | id/title/description/status/createdAt 全有 | **精简到 5 个字段**（type/priority/message/suggestion/related_schedule_ids） |
| 状态枚举 | PENDING/IN_PROGRESS/COMPLETED/CANCELLED | **保留全 4 个**，但增加 `@SerialName` 与后端字符串对齐 |
| 优先级色板 | 红 / 橙黄 / 浅蓝（跨色相） | **蓝色系不同深浅（深蓝→中蓝→浅蓝）**，严格符合"同色系不同深浅"原则 |

## 一、阶段目标（一句话）

> **首页能显示今日时间轴（含冲突分组），能手动新建日程并立即按时间显示出来。**

完成本阶段后，至少能演示这条路径：
`打开 App → 首页看到今日时间轴（含冲突卡片）→ 点击底部"+" → 填写标题/时间/优先级 → 保存 → 首页时间轴立即出现新日程并按时间排序`。

冲突展示（含二级菜单）是第一阶段必交付，不是"第二阶段再做"。

```kotlin
// data/remote/ApiResult.kt

package com.life.app.data.remote

sealed interface ApiResult<out T> {
    data object Loading : ApiResult<Nothing>
    data class Success<T>(val data: T) : ApiResult<T>
    data class Failure(val error: String) : ApiResult<Nothing>
}

fun <T> ApiResponseDto<T>.toApiResult(): ApiResult<T> = when {
    ok && data != null -> ApiResult.Success(data)
    else -> ApiResult.Failure(error ?: "未知错误")
}
```

```kotlin
// domain/model/Schedule.kt

package com.life.app.domain.model

import kotlinx.serialization.SerialName
import kotlinx.serialization.Serializable

@Serializable
enum class Priority {
    @SerialName("P0") P0,
    @SerialName("P1") P1,
    @SerialName("P2") P2
}

@Serializable
enum class ScheduleStatus {
    @SerialName("pending") PENDING,
    @SerialName("in_progress") IN_PROGRESS,
    @SerialName("completed") COMPLETED,
    @SerialName("cancelled") CANCELLED
}

@Serializable
enum class RepeatType {
    @SerialName("none") NONE,
    @SerialName("daily") DAILY,
    @SerialName("weekly") WEEKLY,
    @SerialName("weekdays") WEEKDAYS,
    @SerialName("weekends") WEEKENDS,
    @SerialName("monthly") MONTHLY
}

@Serializable
data class Schedule(
    @SerialName("id") val id: Long,                      // ⚠️ Long，不是 String
    @SerialName("title") val title: String,
    @SerialName("date") val date: String,                // "YYYY-MM-DD"
    @SerialName("start_time") val startTime: String,      // "HH:MM"
    @SerialName("end_time") val endTime: String,         // "HH:MM"
    @SerialName("duration") val durationMinutes: Int,     // ⚠️ 字段名 duration，Kotlin 内部用 durationMinutes
    @SerialName("priority") val priority: Priority,
    @SerialName("status") val status: ScheduleStatus,
    @SerialName("repeat") val repeat: RepeatType = RepeatType.NONE,
    @SerialName("note") val note: String? = null
) {
    val isCompleted: Boolean get() = status == ScheduleStatus.COMPLETED
}
```

```kotlin
object KtorClientFactory {

    /**
     * ⚠️ 端口 8800（与后端规划 §4.1 一致）
     * ⚠️ Android 模拟器访问 host 机器 localhost 是 10.0.2.2，不是 localhost！
     *   真机调试需要 host 和手机在同一局域网，用电脑 IP（开发阶段）
     * ⚠️ 生产环境必须 HTTPS，cleartext 仅限开发阶段（manifest 已配 usesCleartextTraffic）
     *
     * 通过 buildConfigField 注入 BASE_URL，模拟器/真机/生产自动切换：
     *
     *   android {
     *     buildFeatures { buildConfig = true }
     *     buildTypes {
     *       debug {
     *         // 模拟器用 10.0.2.2，真机调试改成电脑局域网 IP（同一 wifi）
     *         buildConfigField("String", "BASE_URL", "\"http://10.0.2.2:8800\"")
     *       }
     *       release {
     *         buildConfigField("String", "BASE_URL", "\"https://api.life.example.com\"")
     *       }
     *     }
     *   }
     */
    fun create(): HttpClient = HttpClient(Android) {
        install(ContentNegotiation) {
            json(Json {
                ignoreUnknownKeys = true
                prettyPrint = false
                encodeDefaults = true
                explicitNulls = false
            })
        }
        install(Logging) {
            level = LogLevel.INFO
        }
        defaultRequest {
            url(BuildConfig.BASE_URL)
            contentType(ContentType.Application.Json)
        }
        expectSuccess = true
    }
}
```

```kotlin
// data/repository/HomeRepository.kt

package com.life.app.data.repository

import com.life.app.data.remote.ApiResult
import com.life.app.data.remote.HomeApi
import com.life.app.data.remote.mapper.toDomain
import com.life.app.data.remote.toApiResult
import com.life.app.domain.model.HomePageData
import com.life.app.domain.model.Reminder
import com.life.app.domain.model.TimelineItem
import javax.inject.Inject

class HomeRepository @Inject constructor(
    private val api: HomeApi
) {
    suspend fun fetchHomeData(date: String): ApiResult<HomePageData> {
        return runCatching {
            val timelineResp = api.getTimeline(date).toApiResult()
            val remindersResp = api.checkReminders(date).toApiResult()

            when {
                timelineResp is ApiResult.Failure -> timelineResp
                remindersResp is ApiResult.Failure -> remindersResp
                timelineResp is ApiResult.Success && remindersResp is ApiResult.Success -> {
                    val timelineData = timelineResp.data
                    val items = timelineData.items.map { it.toDomain() }
                    val topReminder = remindersResp.data
                        .minByOrNull { it.priority }
                        ?.let { dto ->
                            Reminder(
                                type = dto.type,
                                priority = dto.priority,
                                message = dto.message,
                                suggestion = dto.suggestion,
                                relatedScheduleIds = dto.relatedScheduleIds
                            )
                        }
                    ApiResult.Success(
                        HomePageData(
                            date = timelineData.date,
                            weekday = computeWeekday(timelineData.date),  // 第一阶段硬编码
                            timelineItems = items,
                            topReminder = topReminder,
                            empty = items.isEmpty()
                        )
                    )
                }
                else -> ApiResult.Failure("未知状态")
            }
        }.getOrElse { e ->
            ApiResult.Failure(e.message ?: "未知异常")
        }
    }

    private fun computeWeekday(date: String): String {
        // 第一阶段硬编码；后续用 java.time.LocalDate
        return "周四"
    }
}
```

```kotlin
                when (item) {
                    is TimelineItem.Single -> "single-${item.schedule.id}"
                    is TimelineItem.ConflictPair -> "pair-${item.schedules.map { it.id }.sorted().joinToString("-")}"
                    is TimelineItem.ConflictGroup -> "group-${item.schedules.map { it.id }.sorted().joinToString("-")}"
                }
            }
        ) { item ->
            when (item) {
                is TimelineItem.Single -> ScheduleCard(
                    schedule = item.schedule,
                    onClick = onScheduleClick
                )
                is TimelineItem.ConflictPair -> ConflictPairCard(
                    schedules = item.schedules,
                    onScheduleClick = onScheduleClick,
                    onConflictGroupClick = onConflictGroupClick
                )
                is TimelineItem.ConflictGroup -> {
                    var expanded by remember { mutableStateOf(false) }
                    ConflictGroupCard(
                        schedules = item.schedules,
                        expanded = expanded,
                        onExpandToggle = { expanded = !expanded },
                        onScheduleClick = onScheduleClick
                    )
                }
            }
        }
    }
}
```
